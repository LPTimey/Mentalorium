\newglossaryentry{automationbias}
{
name=Automation Bias,
plural=Automation Biases,
description={The tendency to over-rely on automated systems and accept their recommendations despite potential errors}
}

\newglossaryentry{aifallibility}
{
name=AI Fallibility,
plural=AI Fallibilities,
description={The possibility that an AI system may produce incorrect, incomplete, or misleading outputs}
}

\newglossaryentry{mentalhealthapp}
{
name=Mental Health Application,
plural=Mental Health Applications,
description={A software application designed to support mental health assessment, monitoring, or treatment}
}

\newglossaryentry{disclaimer}
{
name=Disclaimer,
plural=Disclaimers,
description={A warning or informational message intended to inform users about limitations or potential inaccuracies of a system}
}

\newglossaryentry{activegroup}
{
name=Active Group,
plural=Active Groups,
description={The experimental group that received disclaimer interventions during Session 1}
}

\newglossaryentry{controlgroup}
{
name=Control Group,
plural=Control Groups,
description={The experimental group that did not receive disclaimer interventions}
}

\newglossaryentry{participant}
{
name=Participant,
plural=Participants,
description={A study subject assigned to an experimental group and two predefined sessions}
}

\newglossaryentry{session}
{
name=Session,
plural=Sessions,
description={A set of three experimental rounds completed by a participant}
}

\newglossaryentry{round}
{
name=Round,
plural=Rounds,
description={A single evaluation consisting of a case vignette and an AI-generated recommendation}
}

\newglossaryentry{casevignette}
{
name=Case Vignette,
plural=Case Vignettes,
description={A short description of a hypothetical mental health case presented to participants}
}

\newglossaryentry{truthfulness}
{
name=Truthfulness,
plural=Truthfulness Labels,
description={A label indicating whether the given AI recommendation is truthful or intentionally untruthful}
}

\newglossaryentry{recommendation}
{
name=Treatment Recommendation,
plural=Treatment Recommendations,
description={A recommendation generated by the application regarding a mental health case}
}

\newglossaryentry{participantallocator}
{
name=Participant Allocator,
plural=Participant Allocators,
description={The deterministic pseudo-random algorithm used to assign participants to valid study configurations}
}

\newglossaryentry{pseudorandomseed}
{
name=Pseudo-random Seed,
plural=Pseudo-random Seeds,
description={A fixed numerical value used to ensure reproducibility of pseudo-random participant allocation}
}

\newglossaryentry{biasscore}
{
name=Automation Bias Score,
plural=Automation Bias Scores,
description={The heuristic score B(i,r,c) used to quantify evidence of automation bias}
}

\newglossaryentry{biasthreshold}
{
name=Bias Threshold,
plural=Bias Thresholds,
description={Our predefined cutoff value above which automation bias is considered present}
}

\newglossaryentry{likert}
{
name=Likert Scale,
plural=Likert Scales,
description={A psychometric response scale commonly used to measure attitudes or opinions}
}

\newglossaryentry{python}
{
name=Python,
plural=Python Programs,

}
